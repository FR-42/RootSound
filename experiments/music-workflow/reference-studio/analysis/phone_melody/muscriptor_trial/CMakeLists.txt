cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(rootsound_muscriptor_short LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MUSCRIPTOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_METAL OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_VULKAN OFF CACHE BOOL "" FORCE)
if(NOT DEFINED MUSCRIPTOR_SOURCE)
  message(FATAL_ERROR "MUSCRIPTOR_SOURCE must point to the pinned upstream cpp folder")
endif()
add_subdirectory("${MUSCRIPTOR_SOURCE}" muscriptor)
add_executable(muscriptor_short trial.cpp)
target_link_libraries(muscriptor_short PRIVATE muscriptor_ggml)
